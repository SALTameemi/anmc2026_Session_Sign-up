-- ===================================================================== -- ANMC 2026 — Session sign-up (attendees pre-book parallel breakout rooms) -- Each room has a capacity (default 30). When a room is full it closes. -- One choice per attendee per time block (changeable). -- Run in the Supabase SQL editor. -- ===================================================================== -- 1) Sign-up table create table if not exists session_signups( id bigserial primary key, registrant_id text not null, attendee_name text, day int not null, -- 2 or 3 block_key text not null, -- e.g. '2-B1' block_label text, room_key text not null, -- e.g. 'Room 2' room_label text, created_at timestamptz default now(), unique(registrant_id, block_key) -- one room per time block ); create index if not exists ix_signup_block_room on session_signups(block_key, room_key); alter table session_signups enable row level security; drop policy if exists ss_sel on session_signups; drop policy if exists ss_ins on session_signups; drop policy if exists ss_upd on session_signups; drop policy if exists ss_del on session_signups; create policy ss_sel on session_signups for select to anon, authenticated using (true); create policy ss_ins on session_signups for insert to anon, authenticated with check (true); create policy ss_upd on session_signups for update to anon, authenticated using (true) with check (true); create policy ss_del on session_signups for delete to anon, authenticated using (true); -- 2) Attendee name list for the picker create or replace function attendee_list() returns json language sql security definer set search_path=public as $$ select coalesce(json_agg(json_build_object('id',registrant_id,'name',full_name) order by full_name),'[]'::json) from registrations where full_name is not null and btrim(full_name) <> ''; $$; grant execute on function attendee_list() to anon, authenticated; -- 2b) Look up an attendee by their registration ID (from the confirmation email) create or replace function lookup_attendee(p_code text) returns json language sql security definer set search_path=public as $$ select coalesce( (select json_build_object('id',registrant_id,'name',full_name) from registrations where upper(btrim(registrant_id)) = upper(btrim(p_code)) limit 1), json_build_object('status','not_found')); $$; grant execute on function lookup_attendee(text) to anon, authenticated; -- 2c) Flexible attendee search: match by registration ID, QID, licence, or name create or replace function find_attendee(q text) returns json language sql security definer set search_path=public as $$ with m as ( select registrant_id, full_name, organization from registrations where nullif(btrim(q),'') is not null and ( upper(btrim(registrant_id)) = upper(btrim(q)) or upper(btrim(coalesce(licence_no,''))) = upper(btrim(q)) or ( regexp_replace(q,'\D','','g') <> '' and regexp_replace(coalesce(qid,''),'\D','','g') = regexp_replace(q,'\D','','g') ) -- phone: match when the typed digits (>=5) appear in the mobile number or ( length(regexp_replace(q,'\D','','g')) >= 5 and regexp_replace(coalesce(mobile,''),'\D','','g') like '%'||regexp_replace(q,'\D','','g')||'%' ) -- name: whitespace-normalised, case-insensitive substring (handles double spaces) or regexp_replace(lower(full_name),'\s+',' ','g') like '%'||regexp_replace(lower(btrim(q)),'\s+',' ','g')||'%' ) order by full_name limit 25 ) select coalesce(json_agg(json_build_object( 'id',registrant_id,'name',full_name,'org',organization)),'[]'::json) from m; $$; grant execute on function find_attendee(text) to anon, authenticated; -- 3) Live counts per block/room -> { "2-B1": {"Room 2": 18, ...}, ... } create or replace function session_counts() returns json language sql security definer set search_path=public as $$ select coalesce(json_object_agg(block_key, rc), '{}'::json) from ( select block_key, json_object_agg(room_key, n) rc from ( select block_key, room_key, count(*)::int n from session_signups group by block_key, room_key ) t group by block_key ) u; $$; grant execute on function session_counts() to anon, authenticated; -- 4) What has this attendee already booked? create or replace function my_sessions(p_reg text) returns json language sql security definer set search_path=public as $$ select coalesce(json_agg(json_build_object( 'block_key',block_key,'room_key',room_key,'room_label',room_label)),'[]'::json) from session_signups where registrant_id = p_reg; $$; grant execute on function my_sessions(text) to anon, authenticated; -- 5) Attendee booking: book ONCE per time block, then LOCKED. -- An attendee cannot change or cancel their own choice — only the -- registration team can (see admin functions below). This prevents chaos. create or replace function book_session( p_reg text, p_name text, p_day int, p_block text, p_block_label text, p_room text, p_room_label text, p_cap int default 30 ) returns json language plpgsql security definer set search_path=public as $$ declare existing text; c int; begin select room_key into existing from session_signups where registrant_id = p_reg and block_key = p_block; if existing is not null then if existing = p_room then return json_build_object('status','ok','room',p_room); -- same, no-op else return json_build_object('status','locked','room',existing); -- already booked, can't self-change end if; end if; -- seats already taken in the target room select count(*)::int into c from session_signups where block_key = p_block and room_key = p_room; if c >= p_cap then return json_build_object('status','full','room',p_room,'count',c); end if; insert into session_signups(registrant_id,attendee_name,day,block_key,block_label,room_key,room_label) values (p_reg,p_name,p_day,p_block,p_block_label,p_room,p_room_label) on conflict (registrant_id, block_key) do nothing; return json_build_object('status','ok','room',p_room,'count',c+1); end; $$; grant execute on function book_session(text,text,int,text,text,text,text,int) to anon, authenticated; -- 6a) ADMIN: move/change an attendee to another room for a block (team only). create or replace function admin_move_session( p_reg text, p_name text, p_day int, p_block text, p_block_label text, p_room text, p_room_label text, p_cap int default 30 ) returns json language plpgsql security definer set search_path=public as $$ declare existing text; c int; begin select room_key into existing from session_signups where registrant_id = p_reg and block_key = p_block; if existing is not distinct from p_room then return json_build_object('status','ok','room',p_room); end if; select count(*)::int into c from session_signups where block_key = p_block and room_key = p_room and registrant_id <> p_reg; if c >= p_cap then return json_build_object('status','full','room',p_room,'count',c); end if; insert into session_signups(registrant_id,attendee_name,day,block_key,block_label,room_key,room_label) values (p_reg,p_name,p_day,p_block,p_block_label,p_room,p_room_label) on conflict (registrant_id, block_key) do update set room_key=excluded.room_key, room_label=excluded.room_label, day=excluded.day, block_label=excluded.block_label, created_at=now(); return json_build_object('status','ok','room',p_room,'count',c+1); end; $$; grant execute on function admin_move_session(text,text,int,text,text,text,text,int) to anon, authenticated; -- 6b) ADMIN: cancel/remove a booking for a block (team only) create or replace function cancel_session(p_reg text, p_block text) returns json language sql security definer set search_path=public as $$ delete from session_signups where registrant_id=p_reg and block_key=p_block; select json_build_object('status','ok'); $$; grant execute on function cancel_session(text,text) to anon, authenticated; notify pgrst, 'reload schema'; -- Maintenance: -- Clear all sign-ups: truncate table session_signups restart identity;
